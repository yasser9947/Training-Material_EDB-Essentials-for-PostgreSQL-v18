DDL, DML, DCL, TCLDDL, DML, DCL, TCL

-- DDL: Data Definition Language (schema)
CREATE TABLE employees (
  id          SERIAL PRIMARY KEY,
  name        VARCHAR(100) NOT NULL,
  email       VARCHAR(255) UNIQUE,
  salary      NUMERIC(10,2) CHECK (salary > 0),
  hire_date   DATE DEFAULT CURRENT_DATE,
  dept_id     INT REFERENCES departments(id)
);

ALTER TABLE employees ADD COLUMN phone VARCHAR(20);
DROP TABLE employees;

-- DML: Data Manipulation Language
INSERT INTO employees (name, email, salary)
VALUES ('Alice', 'alice@example.com', 50000);

UPDATE employees SET salary = salary * 1.10 WHERE dept_id = 5;

DELETE FROM employees WHERE hire_date < '2020-01-01';

SELECT e.name, d.dept_name
FROM employees e
JOIN departments d ON d.id = e.dept_id
WHERE e.salary > 60000
ORDER BY e.name;

-- DCL: Data Control Language
GRANT SELECT ON employees TO analyst;
REVOKE UPDATE ON employees FROM alice;

-- TCL: Transaction Control Language
BEGIN;
  UPDATE accounts SET balance = balance - 100 WHERE id = 1;
  SAVEPOINT before_credit;
  UPDATE accounts SET balance = balance + 100 WHERE id = 2;
  -- oops, wrong account — rollback partial
  -- ROLLBACK TO before_credit;
COMMIT;

-- Customers with their orders
SELECT c.name, o.order_date, o.total
FROM customers c
LEFT JOIN orders o ON o.customer_id = c.id
ORDER BY c.name, o.order_date;

-- Customers who never ordered
SELECT c.name
FROM customers c
LEFT JOIN orders o ON o.customer_id = c.id
WHERE o.id IS NULL;

Indexes و ExplainIndexes & Explain

-- Create indexes
CREATE INDEX idx_orders_customer ON orders(customer_id);
CREATE UNIQUE INDEX idx_users_email ON users(email);
CREATE INDEX idx_orders_date_brin ON orders USING BRIN(order_date);
CREATE INDEX idx_products_data ON products USING GIN(metadata);

-- Partial index (only recent rows)
CREATE INDEX idx_orders_recent ON orders(customer_id)
WHERE order_date > '2024-01-01';

-- Analyze query plan
EXPLAIN SELECT * FROM orders WHERE customer_id = 42;

-- Actual execution with timings
EXPLAIN (ANALYZE, BUFFERS)
SELECT c.name, SUM(o.total) FROM customers c
JOIN orders o ON o.customer_id = c.id
GROUP BY c.name;
