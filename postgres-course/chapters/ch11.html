DDL — تعريف الهيكلDDL — Structure Definition

-- Create table with constraints
CREATE TABLE employees (
  id           SERIAL PRIMARY KEY,                       -- auto-increment PK
  name         TEXT NOT NULL,
  email        TEXT UNIQUE NOT NULL,
  salary       NUMERIC(10,2) CHECK (salary > 0),
  hire_date    DATE DEFAULT CURRENT_DATE,
  dept_id      INT REFERENCES departments(id) ON DELETE SET NULL,
  status       TEXT DEFAULT 'active'
               CHECK (status IN ('active','inactive','on_leave')),
  metadata     JSONB,
  created_at   TIMESTAMPTZ DEFAULT NOW()
);

-- Alter table
ALTER TABLE employees ADD COLUMN phone TEXT;
ALTER TABLE employees DROP COLUMN phone;
ALTER TABLE employees RENAME COLUMN name TO full_name;
ALTER TABLE employees ALTER COLUMN salary TYPE NUMERIC(12,2);
ALTER TABLE employees ADD CONSTRAINT chk_email CHECK (email LIKE '%@%');

-- Add foreign key after creation
ALTER TABLE employees ADD CONSTRAINT fk_dept
  FOREIGN KEY (dept_id) REFERENCES departments(id);

-- Drop table (careful!)
DROP TABLE employees;
DROP TABLE IF EXISTS employees CASCADE;   -- also drops dependencies

-- Rename
ALTER TABLE employees RENAME TO staff;

Transactions و SavepointsTransactions & Savepoints

-- Basic transaction
BEGIN;
  UPDATE accounts SET balance = balance - 100 WHERE id = 1;
  UPDATE accounts SET balance = balance + 100 WHERE id = 2;
COMMIT;  -- or ROLLBACK;

-- With savepoints (partial rollback)
BEGIN;
  INSERT INTO orders (customer_id, total) VALUES (42, 200);
  SAVEPOINT after_order;

  INSERT INTO order_items VALUES (...);
  -- oh no, wrong data
  ROLLBACK TO SAVEPOINT after_order;   -- undo items but keep order

  INSERT INTO order_items VALUES (...);  -- correct this time
COMMIT;

-- Isolation levels (default: READ COMMITTED)
BEGIN ISOLATION LEVEL REPEATABLE READ;
  -- inside this tx, snapshots are consistent
COMMIT;

BEGIN ISOLATION LEVEL SERIALIZABLE;
  -- strongest — like being alone with the DB
COMMIT;

-- Read-only transaction (safer, faster)
BEGIN READ ONLY;
  SELECT ...;
COMMIT;

Views و Materialized ViewsViews & Materialized Views

VIEW

استعلام محفوظ — يُنفذ في كل مرة.

Saved query — re-executes each time.

  • ✓ دائماً بيانات محدثة
  • ✓ Always fresh data
  • ✗ بطيء للاستعلامات المعقدة
  • ✗ Slow for complex queries

MATERIALIZED VIEW

جدول فعلي مبني على استعلام — يُحدّث يدوياً.

Physical table from a query — refreshed manually.

  • ✓ سريع جداً
  • ✓ Very fast
  • ✗ بيانات قد تكون قديمة
  • ✗ Data may be stale
-- Regular VIEW
CREATE VIEW active_customers AS
SELECT c.*, COUNT(o.id) AS order_count
FROM customers c
LEFT JOIN orders o ON o.customer_id = c.id
WHERE c.status = 'active'
GROUP BY c.id;

SELECT * FROM active_customers;

-- MATERIALIZED VIEW
CREATE MATERIALIZED VIEW sales_summary AS
SELECT DATE_TRUNC('day', order_date) AS day,
       COUNT(*) AS orders,
       SUM(total) AS revenue
FROM orders
GROUP BY day
WITH DATA;

-- Create index on it (for speed)
CREATE UNIQUE INDEX ON sales_summary(day);

-- Refresh when data changes
REFRESH MATERIALIZED VIEW sales_summary;

-- Refresh without blocking reads (needs unique index!)
REFRESH MATERIALIZED VIEW CONCURRENTLY sales_summary;

أنواع الـ JoinsJoin Types

INNER JOIN

فقط الصفوف المطابقة في الجدولين

Only matching rows in both

LEFT OUTER JOIN

كل من اليسار + المطابق من اليمين

All left + matches from right

RIGHT OUTER JOIN

كل من اليمين + المطابق من اليسار

All right + matches from left

FULL OUTER JOIN

كل الصفوف من الجدولين

All rows from both

CROSS JOIN

ضرب ديكارتي — كل تركيبة

Cartesian product — all combinations

LATERAL JOIN

استعلام يستخدم قيم الصف السابق

Subquery uses prior row values

-- INNER JOIN — matching only
SELECT c.name, o.total
FROM customers c
JOIN orders o ON o.customer_id = c.id;

-- LEFT JOIN — include customers without orders
SELECT c.name, COALESCE(SUM(o.total), 0) AS total_spent
FROM customers c
LEFT JOIN orders o ON o.customer_id = c.id
GROUP BY c.name;

-- Customers who NEVER ordered (common pattern)
SELECT c.name
FROM customers c
LEFT JOIN orders o ON o.customer_id = c.id
WHERE o.id IS NULL;

-- LATERAL — for each customer, get their latest 3 orders
SELECT c.name, recent.order_date, recent.total
FROM customers c
CROSS JOIN LATERAL (
  SELECT order_date, total
  FROM orders o
  WHERE o.customer_id = c.id
  ORDER BY order_date DESC
  LIMIT 3
) AS recent;

الفهارس — ٦ أنواعIndexes — 6 Types

TypeالاستخدامUse case
B-Treeالافتراضي — <, =, >، BETWEEN، ORDER BYDefault — <, =, >, BETWEEN, ORDER BY
Hashمساواة فقط (=) — أسرع قليلاً من B-treeEquality only (=) — slightly faster
GINJSONB، arrays، full-text searchJSONB, arrays, full-text search
GiSTبيانات جغرافية، rangesGeospatial, ranges

-- Basic B-tree
CREATE INDEX idx_orders_customer ON orders(customer_id);

-- Unique
CREATE UNIQUE INDEX idx_users_email ON users(email);

-- Multi-column (order matters!)
CREATE INDEX idx_orders_cust_date ON orders(customer_id, order_date);

-- Partial — index only a subset (saves space)
CREATE INDEX idx_orders_recent ON orders(customer_id)
WHERE order_date > '2024-01-01';

-- Expression (functional index)
CREATE INDEX idx_email_lower ON users (LOWER(email));
-- Now WHERE LOWER(email) = 'x' uses the index

-- GIN on JSONB column
CREATE INDEX idx_products_meta ON products USING GIN(metadata);
-- Now WHERE metadata @> '{"category": "book"}' is fast

-- BRIN on huge time-series table
CREATE INDEX idx_logs_time_brin ON logs USING BRIN(created_at);
-- 1000x smaller than B-tree for time-sorted data

-- INCLUDE (covering index) — avoid heap reads
CREATE INDEX idx_orders_cust_inc ON orders(customer_id) INCLUDE (total, order_date);

-- Concurrent build (no locks — for live production)
CREATE INDEX CONCURRENTLY idx_new ON orders(status);

-- Drop
DROP INDEX idx_orders_customer;
DROP INDEX CONCURRENTLY idx_orders_customer;

EXPLAIN — قراءة خطط التنفيذEXPLAIN — Reading Execution Plans

-- EXPLAIN — shows plan without running
EXPLAIN SELECT * FROM orders WHERE customer_id = 42;

-- EXPLAIN ANALYZE — actually runs + real timings
EXPLAIN ANALYZE SELECT * FROM orders WHERE customer_id = 42;

-- Full details
EXPLAIN (ANALYZE, BUFFERS, VERBOSE, FORMAT text)
SELECT c.name, SUM(o.total)
FROM customers c JOIN orders o ON o.customer_id = c.id
WHERE c.created_at > '2024-01-01'
GROUP BY c.name;

-- Common plan node types to recognize:
-- * Seq Scan       — full table read (bad if large table!)
-- * Index Scan     — uses index (good)
-- * Index Only Scan — uses only the index (best!)
-- * Bitmap Heap Scan — intermediate, for multiple indexes
-- * Hash Join      — builds hash table (good for medium tables)
-- * Nested Loop    — good for small outer + indexed inner
-- * Merge Join     — good for pre-sorted inputs
-- * Sort           — explicit sorting (watch memory!)
كيف تقرأ الخطة؟ How to read plans? من الداخل للخارج ومن الأسفل للأعلى. الأرقام: cost=startup..total، rows=توقع، actual rows=الفعلي. إذا الفرق كبير = إحصائيات قديمة (نفذ ANALYZE). Inside-out, bottom-up. Numbers: cost=startup..total, rows=estimate, actual rows=real. Big gap = stale stats (run ANALYZE).
