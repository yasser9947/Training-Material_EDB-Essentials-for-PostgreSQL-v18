pg_dump و pg_restorepg_dump & pg_restore

# Plain SQL dump (human-readable)
pg_dump -U postgres -d mydb -f mydb.sql

# Custom format (compressed, allows selective restore)
pg_dump -U postgres -d mydb -F c -f mydb.dump

# Directory format (parallel dump)
pg_dump -U postgres -d mydb -F d -j 4 -f mydb_dir/

# Tar format
pg_dump -U postgres -d mydb -F t -f mydb.tar

# Dump only schema (no data)
pg_dump -U postgres -d mydb --schema-only -f schema.sql

# Dump only data
pg_dump -U postgres -d mydb --data-only -f data.sql

# Dump specific tables
pg_dump -U postgres -d mydb -t customers -t orders -f tables.sql

# All databases + global objects (roles, tablespaces)
pg_dumpall -U postgres -f all.sql

# Restore
psql -U postgres -d mydb -f mydb.sql
pg_restore -U postgres -d mydb mydb.dump
pg_restore -U postgres -d mydb -j 4 mydb_dir/  # parallel

pg_basebackup و PITRpg_basebackup & PITR

PITR يتيح استعادة قاعدة البيانات إلى أي نقطة زمنية — يتطلب physical backup + WAL archiving مفعل.

PITR lets you restore to any point in time — requires physical backup + WAL archiving enabled.

# 1. Enable WAL archiving in postgresql.conf
# wal_level = replica
# archive_mode = on
# archive_command = 'cp %p /archive/%f'

# 2. Take a base backup
pg_basebackup -D /backup/base -Ft -z -P -U replicator

# 3. Simulate a disaster (bad DELETE at 14:30)
# ...

# 4. Restore base backup
rm -rf /var/lib/pgsql/18/data/*
tar -xzf /backup/base/base.tar.gz -C /var/lib/pgsql/18/data/

# 5. Create recovery.signal and configure recovery
touch /var/lib/pgsql/18/data/recovery.signal
cat >> postgresql.auto.conf << EOF
restore_command = 'cp /archive/%f %p'
recovery_target_time = '2026-04-22 14:25:00'
recovery_target_action = 'promote'
EOF

# 6. Start server — it will replay WAL up to the target
pg_ctl start
